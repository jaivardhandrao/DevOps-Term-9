$ docker compose ps
NAME                              IMAGE               SERVICE    STATUS
container-networking-backend-1    alpine:3.20         backend    Up

$ docker compose exec frontend wget -qO- http://backend:8080

$ docker compose exec backend nc -zvw 3 database 3306

$ docker compose exec frontend sh -c 'getent hosts database || echo isolated'
isolated

$ docker network ls --filter name=container-networking --format '{{.Name}}'

container-networking_database_net
container-networking_frontend_net

$ docker inspect container-networking-backend-1 \
    --format '{{range $name, $config := .NetworkSettings.Networks}}{{$name}}{{"\n"}}{{end}}'
container-networking_application_net
container-networking_database_net

Result: frontend can reach backend, backend can reach MySQL, and frontend is isolated from MySQL.
